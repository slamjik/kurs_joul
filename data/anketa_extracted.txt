<script>
// ══════════════════════════════════════════════════════════════════
// ДАННЫЕ: список преподавателей (id → отображаемое имя)
// При добавлении нового преподавателя — добавить сюда и в HTML
// ══════════════════════════════════════════════════════════════════
const TEACHERS = {
    t1: 'Иванова И.И. (Высшая математика)',
    t2: 'Петров П.П. (Физика)',
    t3: 'Сидорова С.С. (Информатика)'
};
 
// Блоки (b1..b5)
const BLOCKS = ['b1','b2','b3','b4','b5'];
 
// Хранилище выборов: choices[teacherId][blockId] = 'fully'|'partially'|'not'|null
const choices = {};
Object.keys(TEACHERS).forEach(tid => {
    choices[tid] = {};
    BLOCKS.forEach(bid => choices[tid][bid] = null);
});
 
let isOpenQuestionsStage = false;
 
// ── Клик по кнопке выбора ──────────────────────────────────────
function selectOpt(btn, teacherId, blockId) {
    const val = btn.getAttribute('data-val');
 
    // Снять активный класс со всех кнопок этой строки (той же пары teacher+block)
    const row = btn.closest('tr');
    row.querySelectorAll('.btn-opt').forEach(b => b.classList.remove('active'));
    btn.classList.add('active');
 
    // Сохранить выбор
    choices[teacherId][blockId] = val;
 
    // Управление строкой-раскрывашкой
    const expandRow = document.getElementById(`expand-${teacherId}-${blockId}`);
    if (val === 'partially') {
        expandRow.classList.add('show');
    } else {
        expandRow.classList.remove('show');
        expandRow.querySelectorAll('input[type="checkbox"]').forEach(cb => cb.checked = false);
    }
}
 
// ── Валидация и переход к открытым вопросам ───────────────────
function validateAndNext() {
 
    if (!isOpenQuestionsStage) {
        // ЭТАП 1: проверить, что все ячейки заполнены
        for (const tid of Object.keys(TEACHERS)) {
            for (const bid of BLOCKS) {
                if (!choices[tid][bid]) {
                    alert(`Пожалуйста, выберите оценку для преподавателя ${TEACHERS[tid]} в блоке ${bid.replace('b','Блок ')}.`);
                    // Прокрутить к нужному блоку
                    const blockNum = bid.replace('b','');
                    document.querySelector(`[data-block-id="${blockNum}"]`).scrollIntoView({behavior:'smooth', block:'center'});
                    return;
                }
                // Если «Частично» — обязательно хотя бы одна галочка
                if (choices[tid][bid] === 'partially') {
                    const expandRow = document.getElementById(`expand-${tid}-${bid}`);
                    const checked = expandRow.querySelectorAll('input[type="checkbox"]:checked').length;
                    if (checked === 0) {
                        alert(`Вы выбрали «Частично» для ${TEACHERS[tid]} (${bid.replace('b','Блок ')}). Отметьте хотя бы одну галочку в раскрывшемся списке!`);
                        expandRow.scrollIntoView({behavior:'smooth', block:'center'});
                        return;
                    }
                }
            }
        }
 
        // Определить преподавателей с хотя бы одним «Частично» или «Не сделано»
        const problemTeachers = Object.keys(TEACHERS).filter(tid =>
            BLOCKS.some(bid => choices[tid][bid] === 'partially' || choices[tid][bid] === 'not')
        );
 
        // Сгенерировать текстовые поля
        const container = document.getElementById('dynamicTextareasContainer');
        container.innerHTML = '';
 
        if (problemTeachers.length > 0) {
            problemTeachers.forEach(tid => {
                const group = document.createElement('div');
                group.className = 'text-area-group';
                group.innerHTML = `
                    <label style="color:var(--danger);">⚠️ ${TEACHERS[tid]}</label>
                    <p style="font-size:12px;color:#64748b;margin-top:-4px;margin-bottom:8px;">
                        Что именно вас не устраивает в организации дисциплины или подаче материала?
                        Напишите подробно (какие темы непонятны, чего не хватает, что мешает учёбе):
                    </p>
                    <textarea class="required-comment" data-teacher="${TEACHERS[tid]}" placeholder="Опишите ваши замечания подробно..."></textarea>
                `;
                container.appendChild(group);
            });
        } else {
            container.innerHTML = '<p style="font-size:14px;color:var(--success);font-weight:600;">🎉 Отлично! Вы полностью довольны организацией учебного процесса по всем предметам.</p>';
        }
 
        // Скрыть блоки, показать открытые вопросы
        document.querySelectorAll('.block-card').forEach(c => c.style.display = 'none');
        document.getElementById('openQuestionsSection').style.display = 'block';
        document.getElementById('mainSubmitBtn').innerText = 'Отправить анкету в базу данных';
        isOpenQuestionsStage = true;
        window.scrollTo(0, 0);
 
    } else {
        // ЭТАП 2: проверить заполненность обязательных текстовых полей
        for (const ta of document.querySelectorAll('.required-comment')) {
            if (ta.value.trim() === '') {
                alert(`Пожалуйста, заполните текстовое описание проблемы по дисциплине: ${ta.getAttribute('data-teacher')}`);
                ta.focus();
                return;
            }
        }
 
        // Успешная отправка
        alert('Анкета успешно сохранена! Большое спасибо за участие в оценке качества образования.');
        // Здесь подключается Ajax/Fetch-запрос к бэкенду
        document.getElementById('quizForm').reset();
        location.reload();
    }
}
</script>
